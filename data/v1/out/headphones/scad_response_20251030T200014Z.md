// Headphones model

// Headband (curved cylinder)
color([0.133, 0.133, 0.133]) // #222222
rotate([0, 0, 90])
translate([0, 0.8, 0])
cylinder(h=2.5, r=0.1, $fn=16);

// Left ear cup
color([0.2, 0.2, 0.2]) // #333333
translate([-1.3, 0.8, 0])
cylinder(h=0.2, r=0.5, $fn=32);

// Right ear cup
color([0.2, 0.2, 0.2]) // #333333
translate([1.3, 0.8, 0])
cylinder(h=0.2, r=0.5, $fn=32);

// Left ear cushion (approximated with torus using difference of cylinders)
color([0.067, 0.067, 0.067]) // #111111
translate([-1.3, 0.8, 0.1])
rotate([90, 0, 0])
difference() {
    cylinder(h=0.05, r=0.45, $fn=32);
    cylinder(h=0.05, r=0.4, $fn=32);
}

// Right ear cushion (approximated with torus using difference of cylinders)
color([0.067, 0.067, 0.067]) // #111111
translate([1.3, 0.8, 0.1])
rotate([90, 0, 0])
difference() {
    cylinder(h=0.05, r=0.45, $fn=32);
    cylinder(h=0.05, r=0.4, $fn=32);
}

// Left connector
color([0.133, 0.133, 0.133]) // #222222
