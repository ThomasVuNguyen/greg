```javascript
import * as THREE from 'https://cdn.jsdelivr.net/npm/three@0.132.2/build/three.module.js';
import { OrbitControls } from 'https://cdn.jsdelivr.net/npm/three@0.132.2/examples/jsm/controls/OrbitControls.js';

const scene = new THREE.Scene();
scene.background = new THREE.Color(0xf0f0f0);

const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
camera.position.set(0, 2, 5);

const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setSize(window.innerWidth, window.innerHeight);
document.body.appendChild(renderer.domElement);

const controls = new OrbitControls(camera, renderer.domElement);
controls.enableDamping = true;

const ambientLight = new THREE.AmbientLight(0x404040, 0.6);
scene.add(ambientLight);

const directionalLight = new THREE.DirectionalLight(0xffffff, 0.8);
directionalLight.position.set(5, 10, 7);
scene.add(directionalLight);

const bodyGeometry = new THREE.SphereGeometry(0.8, 32, 32);
const bodyMaterial = new THREE.MeshPhongMaterial({ color: 0x666666 });
const body = new THREE.Mesh(bodyGeometry, bodyMaterial);
body.position.y = 0.5;
scene.add(body);

const headGeometry = new THREE.SphereGeometry(0.5, 32, 32);
const headMaterial = new THREE.MeshPhongMaterial({ color: 0x888888 });
const head = new THREE.Mesh(headGeometry, headMaterial);
head.position.set(0, 1.2, 0.7);
scene.add(head);

const earGeometry = new THREE.ConeGeometry(0.2, 0.4, 8);
const earMaterial = new THREE.MeshPhongMaterial({ color: 0x888888 });

const leftEar = new THREE.Mesh(earGeometry, earMaterial);
leftEar.position.set(-0.3, 1.6, 0.5);
leftEar.rotation.x = Math.PI / 4;
scene.add(leftEar);

const rightEar = new THREE.Mesh(earGeometry, earMaterial);
rightEar.position.set(0.3, 1.6, 0.5);
rightEar.rotation.x = Math.PI / 4;
scene.add(rightEar);

const legGeometry = new THREE.CylinderGeometry(0.1, 0.1, 0.4, 12);
const legMaterial = new THREE.MeshPhongMaterial({ color: 0x666666 });

const frontLeftLeg = new THREE.Mesh(legGeometry, legMaterial);
frontLeftLeg.position.set(-0.3, 0.2, 0.4);
scene.add(frontLeftLeg);

const frontRightLeg = new THREE.Mesh(legGeometry, legMaterial);
frontRightLeg.position.set(0.3, 0.2, 0.4);
scene.add(frontRightLeg);

const backLeftLeg = new THREE.Mesh(legGeometry, legMaterial);
backLeftLeg.position.set(-0.3, 0.2, -0.4);
scene.add(backLeftLeg);

const backRightLeg = new THREE.Mesh(legGeometry, legMaterial);
backRightLeg.position.set(0.3, 0.2, -0.4);
scene.add(backRightLeg);

const tailGeometry = new THREE.CylinderGeometry(0.08, 0.05, 1, 12);
const tailMaterial = new THREE.MeshPhongMaterial({ color: 0x666666 });
const tail = new THREE.Mesh(tailGeometry, tailMaterial);
tail.position.set(0, 0.5, -0.8);
tail.rotation.x = Math.PI / 3;
scene.add(tail);

function animate() {
    requestAnimationFrame(animate);
    controls.update();
    renderer.render(scene, camera);
}

animate();

window.addEventListener('resize', () => {
    camera.aspect = window.innerWidth / window.innerHeight;
    camera.updateProjectionMatrix();
    renderer.setSize(window.innerWidth, window.innerHeight);
});
```
